# Per-host official-repo extras for the host named `sohrab`
# (Intel NUC, Intel integrated graphics / i915).
# Appended to shared/pacman.txt — see hosts/README.md.

# Gaming — lives in the [multilib] repo, which install.sh enables for you.
# `steam` depends on the virtual packages vulkan-driver, lib32-vulkan-driver and
# lib32-libgl, each with several providers. Name this host's GPU drivers
# explicitly in packages/pacman.<hostname>.txt so they are in the same pacman
# transaction — otherwise `--noconfirm` silently picks the first provider, which
# is usually another vendor's driver:
#   Intel  -> vulkan-intel  lib32-mesa lib32-vulkan-intel
#   AMD    -> vulkan-radeon lib32-mesa lib32-vulkan-radeon
#   NVIDIA -> nvidia-utils  lib32-nvidia-utils
steam
vulkan-icd-loader
lib32-vulkan-icd-loader

# Vulkan drivers for this GPU, 64-bit and 32-bit. `steam` depends on the
# virtual packages vulkan-driver / lib32-vulkan-driver / lib32-libgl, each of
# which has several providers; naming Intel's explicitly here puts them in the
# same pacman transaction, so --noconfirm never picks another vendor's driver.
vulkan-intel
lib32-mesa
lib32-vulkan-intel

# CPU microcode — vendor-specific, so it belongs here rather than in the shared
# list (an AMD host wants amd-ucode). The bootloader entry needs it as an extra
# initrd; systemd-boot's pacman hooks handle that automatically.
intel-ucode

# Hardware video acceleration for this i915 GPU. intel-media-driver is the
# modern VA-API driver (Gen8+); libva-intel-driver covers older codepaths some
# applications still request.
intel-media-driver
libva-intel-driver
