# Per-host official-repo extras for the host named `sohrab`
# (Intel NUC, Intel integrated graphics / i915).
# Appended to shared/pacman.txt — see hosts/README.md.
#
# No gaming on this box: `steam` and the 32-bit Vulkan/mesa stack it drags in
# live in hosts/rostam/pacman.txt and hosts/giv/pacman.txt instead. Dropping
# them here does not uninstall anything — ./bin/pkg-diff.sh will report them as
# installed-but-undeclared until they are removed by hand (see MAINTENANCE.md).

# ── GPU: Intel integrated (i915) ─────────────────────────────────────────────
# 64-bit Vulkan only; nothing on this host needs the lib32 halves. Declared
# explicitly rather than left to a virtual `vulkan-driver` dependency so
# --noconfirm can never resolve it to another vendor's driver.
vulkan-intel
vulkan-icd-loader

# CPU microcode — vendor-specific, so it belongs here rather than in the shared
# list (an AMD host wants amd-ucode). The bootloader entry needs it as an extra
# initrd; systemd-boot's pacman hooks handle that automatically.
intel-ucode

# Hardware video acceleration for this i915 GPU. intel-media-driver is the
# modern VA-API driver (Gen8+); libva-intel-driver covers older codepaths some
# applications still request.
intel-media-driver
libva-intel-driver
