# AUR packages — shared across ALL hosts.
# One package per line. Blank lines and #comments are ignored.
# Per-host extras go in  packages/aur.<hostname>.txt

# Terminal
# Built from git, not the official `wezterm` package: the 20240203 stable
# release starts but never maps a Wayland window under Hyprland 0.56 (the
# process stays alive, no window, no error logged). Master fixed it.
wezterm-git

# Fonts
# Persian font. The old `ttf-vazir` was dropped from the AUR — upstream renamed
# the project to Vazirmatn, and this is its package.
vazirmatn-fonts

# Hypr ecosystem extras
# hyprsunset, hyprshot and hyprpolkitagent used to live here; all three moved
# into [extra] and were deleted from the AUR, so they are in pacman.txt now.
wlogout

# GTK/cursor theme (referenced by ~/.config/gtk-*/settings.ini)
bibata-cursor-theme

# App launcher + waybar-with-cava.
# waybar-cava replaces the official `waybar` package (it declares
# conflicts/provides waybar), which is why waybar is NOT in pacman.txt: pacman
# will not remove a conflicting installed package under --noconfirm.
walker-bin
waybar-cava
libcava

# Rootless Docker. `dockerd-rootless-setuptool.sh` is NOT part of Arch's
# `docker` package — it only ships here, and install.sh skips the whole rootless
# setup (leaving Docker unusable) when this is missing.
docker-rootless-extras

# IDEs / agents
claude-code
visual-studio-code-bin
cursor-bin

# Work / social / media
postman-bin
teams-for-linux
zoom
slack-desktop
jellyfin-media-player
spotify
google-chrome

# Editor tooling (LSP / formatters not in official repos).
# `stylua` moved to [extra] and left the AUR; it is in pacman.txt now.
hyprls

# Python tool
arxiv-latex-cleaner
